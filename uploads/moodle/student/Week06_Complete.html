2. Setup & Feature Selection / الإعداد الأوّلي واختيار المتغيرات المفسِّرة

import pandas as pd
import numpy as np
from sklearn.model_selection import train_test_split
from sklearn.linear_model import LinearRegression
from sklearn.ensemble import RandomForestRegressor
from sklearn.metrics import r2_score, mean_absolute_error, mean_squared_error
import matplotlib.pyplot as plt

df = pd.read_csv('algeria_economic_combined.csv').dropna()

# Features (X) and target (y)
features = ['Inflation_Percent', 'Unemployment_Rate', 'Oil_Revenue_Percent']
target   = 'GDP_Growth_Percent'

X = df[features]
y = df[target]

print(f"Dataset: {X.shape[0]} rows, {X.shape[1]} features")

3. Feature Scaling with StandardScaler / توحيد مقياس المتغيرات المفسِّرة

قبل تدريب النموذج، يُوصى بتوحيد مقياس المتغيرات المفسِّرة (Feature Scaling) حتى لا يهيمن متغير ذو وحدات كبيرة على النموذج. يقوم StandardScaler بتحويل كل عمود ليكون وسطه الحسابي يساوي صفراً وانحرافه المعياري يساوي واحداً — وهو ما يُعرف بالتوحيد القياسي (Standardization).
from sklearn.preprocessing import StandardScaler

scaler = StandardScaler()
X_scaled = scaler.fit_transform(X)   # fit on training data only in practice

4. Train / Test Split / تقسيم البيانات إلى مجموعتَي التدريب والاختبار

# 80% train, 20% test — random_state for reproducibility
X_train, X_test, y_train, y_test = train_test_split(
    X_scaled, y, test_size=0.2, random_state=42
)
print(f"Train size: {len(X_train)}  |  Test size: {len(X_test)}")
نقسّم مجموعة البيانات: 80% لمجموعة التدريب و20% لمجموعة الاختبار. المعامل random_state=42 يضمن قابلية استنساخ النتائج. ملاحظة أساسية: نطبّق fit_transform على مجموعة التدريب فقط، ثم transform فقط على مجموعة الاختبار — لمنع تسرب المعلومات (Data Leakage).

5. Linear Regression (Core) / الانحدار الخطي — الأساسي

lr = LinearRegression()
lr.fit(X_train, y_train)
y_pred_lr = lr.predict(X_test)

# Metrics / المقاييس
r2_lr  = r2_score(y_test, y_pred_lr)
mae_lr = mean_absolute_error(y_test, y_pred_lr)
rmse_lr= np.sqrt(mean_squared_error(y_test, y_pred_lr))

print(f"Linear Regression:")
print(f"  R²   = {r2_lr:.3f}")
print(f"  MAE  = {mae_lr:.3f}")
print(f"  RMSE = {rmse_lr:.3f}")

6. Random Forest (Advanced / Optional) / الغابة العشوائية — نموذج متقدم (اختياري)

الغابة العشوائية نموذج تعلم آلي متقدم — يُنصح بتعلّمه بعد إتقان الانحدار الخطي. يعتمد على بناء مئات من أشجار القرار ثم تجميع تنبؤاتها، مما يجعله أكثر دقة لكن أصعب في التفسير الاقتصادي المباشر مقارنة بالانحدار الخطي.
rf = RandomForestRegressor(n_estimators=100, random_state=42)
rf.fit(X_train, y_train)
y_pred_rf = rf.predict(X_test)

r2_rf  = r2_score(y_test, y_pred_rf)
mae_rf = mean_absolute_error(y_test, y_pred_rf)
rmse_rf= np.sqrt(mean_squared_error(y_test, y_pred_rf))

print(f"Random Forest:")
print(f"  R²   = {r2_rf:.3f}")
print(f"  MAE  = {mae_rf:.3f}")
print(f"  RMSE = {rmse_rf:.3f}")

# Feature importance / أهمية المتغيرات
importance = pd.Series(rf.feature_importances_, index=features).sort_values(ascending=True)
importance.plot(kind='barh', title='Feature Importance / أهمية المتغيرات')

7. Metrics Reference / مرجع المقاييس

MetricFormulaInterpretation / التفسير
R²1 − SS_res/SS_tot0–1; closer to 1 is better / كلما اقترب من 1 كان أفضل
MAEmean(|y − ŷ|)Average error in original units / متوسط الخطأ بالوحدات الأصلية
RMSE√mean((y − ŷ)²)Penalizes large errors / يعاقب الأخطاء الكبيرة

8. Actual vs Predicted Plot / مخطط القيم الفعلية مقابل القيم المتنبَّأ بها

fig, axes = plt.subplots(1, 2, figsize=(13, 5))

for ax, y_pred, name, color in zip(
        axes,
        [y_pred_lr, y_pred_rf],
        ['Linear Regression', 'Random Forest'],
        ['#2E86AB', '#6a1b9a']):

    ax.scatter(y_test, y_pred, color=color, alpha=0.7)
    mn = min(y_test.min(), y_pred.min())
    mx = max(y_test.max(), y_pred.max())
    ax.plot([mn, mx], [mn, mx], 'r--', linewidth=1.5)   # perfect fit line
    ax.set_title(f'{name} — Actual vs Predicted')
    ax.set_xlabel('Actual / الفعلي')
    ax.set_ylabel('Predicted / المتوقع')

plt.tight_layout()
plt.savefig('actual_vs_predicted.png', dpi=150)
plt.show()
النقاط القريبة من الخط الأحمر المنقط (خط التطابق التام) تعني أن القيم المتنبَّأ بها قريبة من القيم الفعلية — أي أن النموذج دقيق. الانتشار الواسع للنقاط حول الخط يدل على ضعف القدرة التنبؤية للنموذج.
