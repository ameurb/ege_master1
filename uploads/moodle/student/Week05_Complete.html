1. Descriptive Statistics Beyond describe() / مقاييس الإحصاء الوصفي المتقدمة

import pandas as pd
import numpy as np
from scipy import stats

df = pd.read_csv('algeria_economic_combined.csv').sort_values('Year')

# Basic describe
df[['GDP_Growth_Percent','Inflation_Percent','Unemployment_Rate']].describe()

# Additional measures
print("Skewness / الالتواء:")
print(df[['GDP_Growth_Percent','Inflation_Percent']].skew())

print("\nKurtosis / التفرطح:")
print(df[['GDP_Growth_Percent','Inflation_Percent']].kurtosis())
StatisticInterpretation / التفسير
skewness > 0Right-skewed — tail to the right / التوزيع ملتوٍ يميناً
skewness < 0Left-skewed — tail to the left / التوزيع ملتوٍ يساراً
skewness ≈ 0Symmetric / متماثل
kurtosis > 3Heavy tails / ذيول ثقيلة
kurtosis ≈ 3Normal distribution / توزيع طبيعي

3. OLS Regression with statsmodels / الانحدار الخطي بطريقة المربعات الصغرى العادية

Unemployment = β₀ + β₁·GDP_Growth + β₂·Inflation + ε
نموذج الانحدار الخطي المتعدد: نستخدم معدل البطالة كمتغير تابع (Y)، ومعدل نمو الناتج المحلي ومعدل التضخم كمتغيرَين مستقلَّين (X) — يتوافق هذا النموذج مع قانون أوكن (Okun's Law) الذي يُفيد بوجود علاقة عكسية بين نمو الناتج والبطالة. β₀ هو الثابت وε هو حد الخطأ العشوائي.
import statsmodels.api as sm

# Dependent variable (Y) / المتغير التابع
Y = df['Unemployment_Rate'].dropna()

# Independent variables (X) / المتغيرات المستقلة
X = df.loc[Y.index, ['GDP_Growth_Percent', 'Inflation_Percent']].dropna()
Y = Y.loc[X.index]   # align after dropna / محاذاة بعد حذف القيم المفقودة

# Add constant (intercept β₀) / إضافة الثابت
X = sm.add_constant(X)

# Fit model / تطبيق النموذج
model = sm.OLS(Y, X).fit()
print(model.summary())

5. Residuals & Diagnostics / تشخيص البواقي والتحقق من افتراضات النموذج

residuals = model.resid

fig, axes = plt.subplots(1, 2, figsize=(12, 4))

# Residual plot / مخطط البواقي
axes[0].scatter(model.fittedvalues, residuals, alpha=0.6)
axes[0].axhline(0, color='red', linestyle='--')
axes[0].set_title('Residuals vs Fitted / البواقي مقابل القيم المقدّرة')
axes[0].set_xlabel('Fitted values')
axes[0].set_ylabel('Residuals')

# Q-Q plot
sm.qqplot(residuals, line='s', ax=axes[1])
axes[1].set_title('Q-Q Plot — Normality of Residuals / اختبار الطبيعية')

plt.tight_layout()
plt.savefig('residual_diagnostics.png', dpi=150)
plt.show()

# Durbin-Watson: test for autocorrelation (value near 2 = no autocorrelation)
from statsmodels.stats.stattools import durbin_watson
print(f"Durbin-Watson: {durbin_watson(residuals):.3f}")

6. Interpretation Template / نموذج التفسير

# After running model.summary(), write:
# بعد تشغيل model.summary()، اكتب التفسير:
print(f"R² = {model.rsquared:.3f} → the model explains {model.rsquared*100:.1f}% of unemployment variance.")
print(f"GDP_Growth coefficient = {model.params['GDP_Growth_Percent']:.3f}")
print(f"  → A 1% increase in GDP growth is associated with a {model.params['GDP_Growth_Percent']:.3f}% change in unemployment.")
print(f"  → This {'confirms' if model.params['GDP_Growth_Percent'] < 0 else 'contradicts'} Okun's Law (negative relation expected).")
if model.pvalues['GDP_Growth_Percent'] < 0.05:
    print("  → Statistically significant at 5% level (p < 0.05)")
else:
    print("  → Not statistically significant (p ≥ 0.05)")
