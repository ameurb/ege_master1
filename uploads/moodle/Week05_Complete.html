1. Descriptive Statistics Beyond describe() / الإحصاء الوصفي المتقدم

import pandas as pd
import numpy as np
from scipy import stats

df = pd.read_csv('algeria_economic_combined.csv').sort_values('Year')

# Basic describe
df[['GDP_Growth_Percent','Inflation_Percent','Unemployment_Rate']].describe()

# Additional measures
print("Skewness / الالتواء:")
print(df[['GDP_Growth_Percent','Inflation_Percent']].skew())

print("\nKurtosis / التفرطح:")
print(df[['GDP_Growth_Percent','Inflation_Percent']].kurtosis())
StatisticInterpretation / التفسير
skewness > 0Right-skewed — tail to the right / التوزيع ملتوٍ يميناً
skewness < 0Left-skewed — tail to the left / التوزيع ملتوٍ يساراً
skewness ≈ 0Symmetric / متماثل
kurtosis > 3Heavy tails / ذيول ثقيلة
kurtosis ≈ 3Normal distribution / توزيع طبيعي

2. Correlation Matrix & Heatmap / مصفوفة الارتباط وخريطة الحرارة

import seaborn as sns
import matplotlib.pyplot as plt

cols = ['GDP_Growth_Percent', 'Inflation_Percent', 'Unemployment_Rate', 'Oil_Revenue_Percent']
corr = df[cols].corr()

fig, ax = plt.subplots(figsize=(8, 6))
sns.heatmap(corr, annot=True, fmt='.2f', cmap='RdYlGn',
            center=0, vmin=-1, vmax=1, ax=ax,
            linewidths=0.5)
ax.set_title('Correlation Matrix — Algeria Economic Indicators', fontsize=13, fontweight='bold')
plt.tight_layout()
plt.savefig('correlation_heatmap.png', dpi=150)
plt.show()
قيمة الارتباط بين -1 و+1. القيم القريبة من +1 تعني علاقة موجبة قوية، القريبة من -1 تعني علاقة سالبة قوية، القريبة من 0 تعني ضعف العلاقة.

3. OLS Regression with statsmodels / الانحدار الخطي البسيط

Y = β₀ + β₁X₁ + β₂X₂ + ε
import statsmodels.api as sm

# Dependent variable / المتغير التابع
Y = df['GDP_Growth_Percent'].dropna()

# Independent variables / المتغيرات المستقلة
X = df.loc[Y.index, ['Inflation_Percent', 'Oil_Revenue_Percent']].dropna()
Y = Y.loc[X.index]   # align after dropna

# Add constant (intercept) / إضافة الثابت
X = sm.add_constant(X)

# Fit model / تطبيق النموذج
model = sm.OLS(Y, X).fit()
print(model.summary())

OutputMeaning / المعنى

P>|t|p-value: < 0.05 = statistically significant / قيمة p: أقل من 0.05 = ذو دلالة إحصائية
Prob (F-statistic)Overall model significance / دلالة النموذج الكلية

5. Residuals & Diagnostics / التحقق من افتراضات النموذج

residuals = model.resid

fig, axes = plt.subplots(1, 2, figsize=(12, 4))

# Residual plot / مخطط البواقي
axes[0].scatter(model.fittedvalues, residuals, alpha=0.6)
axes[0].axhline(0, color='red', linestyle='--')
axes[0].set_title('Residuals vs Fitted / البواقي مقابل القيم المقدّرة')
axes[0].set_xlabel('Fitted values')
axes[0].set_ylabel('Residuals')

# Q-Q plot
sm.qqplot(residuals, line='s', ax=axes[1])
axes[1].set_title('Q-Q Plot — Normality of Residuals / اختبار الطبيعية')

plt.tight_layout()
plt.savefig('residual_diagnostics.png', dpi=150)
plt.show()

# Durbin-Watson: test for autocorrelation (value near 2 = no autocorrelation)
from statsmodels.stats.stattools import durbin_watson
print(f"Durbin-Watson: {durbin_watson(residuals):.3f}")

6. Interpretation Template / نموذج التفسير

# After running model.summary(), write:
print(f"R² = {model.rsquared:.3f} → the model explains {model.rsquared*100:.1f}% of GDP growth variance.")
print(f"Inflation coefficient = {model.params['Inflation_Percent']:.3f}")
print(f"  → A 1% increase in inflation is associated with a {model.params['Inflation_Percent']:.3f}% change in GDP growth.")
if model.pvalues['Inflation_Percent'] < 0.05:
    print("  → Statistically significant (p < 0.05)")
else:
    print("  → Not statistically significant (p ≥ 0.05)")
