1. Setup / الإعداد الأوّلي

import matplotlib.pyplot as plt
import pandas as pd

plt.style.use('seaborn-v0_8-whitegrid')  # clean style
df = pd.read_csv('algeria_economic_combined.csv').sort_values('Year')

3. Complete Line Chart Example

fig, ax = plt.subplots(figsize=(10, 5))

ax.plot(df['Year'], df['GDP_Growth_Percent'],
        color='#2E86AB', linewidth=2, marker='o', markersize=4,
        label='GDP Growth')

ax.axhline(0, color='red', linestyle='--', linewidth=0.8, alpha=0.7)
ax.fill_between(df['Year'], df['GDP_Growth_Percent'], 0,
                where=(df['GDP_Growth_Percent'] < 0),
                alpha=0.2, color='red', label='Recession')

ax.set_title('Algeria GDP Growth Rate (%) 2000–2023', fontsize=14, fontweight='bold')
ax.set_xlabel('Year / السنة')
ax.set_ylabel('Growth Rate (%) / معدل النمو')
ax.legend()

plt.tight_layout()
plt.savefig('gdp_growth.png', dpi=150, bbox_inches='tight')
plt.show()

4. 2×2 Dashboard / لوحة المؤشرات 2×2

fig, axes = plt.subplots(2, 2, figsize=(14, 10))
fig.suptitle('Economic Dashboard / لوحة البيانات', fontsize=14, fontweight='bold')

ax1, ax2, ax3, ax4 = axes[0,0], axes[0,1], axes[1,0], axes[1,1]

# Panel 1: Line
ax1.plot(df['Year'], df['GDP_Growth_Percent'])
ax1.set_title('GDP Growth')

# Panel 2: Bar (top 10 inflation)
top10 = df.nlargest(10,'Inflation_Percent').sort_values('Inflation_Percent')
ax2.barh(top10['Year'].astype(str), top10['Inflation_Percent'], color='salmon')
ax2.set_title('Top 10 Inflation Years')

# Panel 3: Scatter
ax3.scatter(df['GDP_Growth_Percent'], df['Inflation_Percent'], alpha=0.6)
ax3.set_title('Growth vs Inflation')

# Panel 4: Histogram
ax4.hist(df['Unemployment_Rate'].dropna(), bins=10, color='steelblue')
ax4.set_title('Unemployment Distribution')

plt.tight_layout()
plt.savefig('dashboard.png', dpi=150)
plt.show()
